rules_version = '2';
// Regole di sicurezza di Bikefly (beta 13.7): da incollare in Firebase → Firestore Database → Regole
service cloud.firestore {
  match /databases/{database}/documents {
    function accesso() { return request.auth != null; }
    function io() { return request.auth.uid; }
    // email verificata (gli account Google lo sono già)
    function verificato() {
      return request.auth.token.email_verified == true
        || request.auth.token.firebase.sign_in_provider == 'google.com';
    }
    function admin() { return accesso() && exists(/databases/$(database)/documents/admin/$(io())); }
    // bloccato dagli admin, oppure in pausa dopo 3 segnalazioni da persone diverse
    function bandito() {
      return exists(/databases/$(database)/documents/banditi/$(io())) || sospeso();
    }
    function sospeso() {
      return exists(/databases/$(database)/documents/segnalati/$(io()))
        && get(/databases/$(database)/documents/segnalati/$(io())).data.get('conteggio', 0) >= 3
        && get(/databases/$(database)/documents/segnalati/$(io())).data.get('riattivato', false) != true;
    }
    // entrato nella beta chiusa con un codice invito (gli admin sempre)
    function ammesso() {
      return admin() || (exists(/databases/$(database)/documents/utenti/$(io()))
        && get(/databases/$(database)/documents/utenti/$(io())).data.get('ammesso', false) == true);
    }
    // chi può scrivere contenuti: dentro, verificato, invitato e non bloccato
    function scrittore() { return accesso() && verificato() && !bandito() && ammesso(); }
    function testo(x, max) { return x is string && x.size() <= max; }
    function fotoOk(x) { return x == null || (x is string && x.size() <= 150000); }
    function cambiati() { return request.resource.data.diff(resource.data).affectedKeys(); }
    function utenteDopo() { return getAfter(/databases/$(database)/documents/utenti/$(io())).data; }

    // Admin e utenti bloccati: si gestiscono dalla console (o dal pannello admin)
    match /admin/{uid} {
      allow read: if accesso() && io() == uid;
    }
    match /banditi/{uid} {
      allow read: if accesso() && io() == uid;
      allow write: if admin();
    }

    // Profili: ognuno scrive solo il suo. I punti non possono saltare di colpo.
    match /utenti/{uid} {
      allow read: if accesso();
      allow delete: if accesso() && io() == uid;
      allow create: if accesso() && io() == uid && profiloValido()
        && request.resource.data.get('punti', 0) <= 1500
        && (request.resource.data.get('ammesso', false) == false || invitoUsato())
        && (request.resource.data.get('mioCodice', null) == null || mioCodiceOk())
        && (request.resource.data.get('creato', null) == null || request.resource.data.creato == request.time);
      allow update: if accesso() && io() == uid && profiloValido()
        && puntiOnesti() && tempoOnesto('ultimaBici', 60) && tempoOnesto('ultimoPost', 30)
        // si entra solo consumando un invito, e non si torna indietro
        && (!('ammesso' in cambiati()) || (request.resource.data.ammesso == true && resource.data.get('ammesso', false) == false && invitoUsato()))
        && (!('invitatoDa' in cambiati()) || ('ammesso' in cambiati()))
        // il proprio codice si crea una volta sola
        && (!('mioCodice' in cambiati()) || (resource.data.get('mioCodice', null) == null && mioCodiceOk()))
        // la data di iscrizione si scrive una volta sola, con l'ora del server
        && (!('creato' in cambiati()) || (resource.data.get('creato', null) == null && request.resource.data.creato == request.time));

      function invitoUsato() {
        let c = request.resource.data.invitatoDa;
        return c is string
          && getAfter(/databases/$(database)/documents/inviti/$(c)).data.usi == get(/databases/$(database)/documents/inviti/$(c)).data.usi + 1;
      }
      function mioCodiceOk() {
        return getAfter(/databases/$(database)/documents/inviti/$(request.resource.data.mioCodice)).data.proprietario == io();
      }

      function profiloValido() {
        let d = request.resource.data;
        return testo(d.get('nome', ''), 30)
          && (d.get('scuola', null) == null || testo(d.scuola, 60))
          && d.get('punti', 0) is int && d.get('punti', 0) >= 0
          && d.get('km', 0) is number && d.get('km', 0) >= 0
          && (d.get('foto', null) == null || (d.foto is string && d.foto.size() <= 150000));
      }
      // al massimo +250 punti e +60 km ogni 10 minuti
      function puntiOnesti() {
        let d = request.resource.data;
        let vecchi = resource.data.get('punti', 0);
        let prima = resource.data.get('aggPunti', timestamp.value(0));
        return d.get('punti', 0) <= vecchi
          || (d.get('punti', 0) - vecchi <= 250
              && d.get('km', 0) - resource.data.get('km', 0) <= 60
              && d.get('aggPunti', null) == request.time
              && request.time > prima + duration.value(10, 'm'));
      }
      // ultimaBici / ultimoPost: solo l'ora del server e con una pausa minima
      function tempoOnesto(campo, secondi) {
        return !(campo in cambiati())
          || (request.resource.data[campo] == request.time
              && request.time > resource.data.get(campo, timestamp.value(0)) + duration.value(secondi, 's'));
      }
    }

    // Codici invito: ognuno ne ha uno con 3 usi, gli admin possono crearne di più grandi
    match /inviti/{codice} {
      allow get: if accesso();
      allow list, delete: if admin();
      allow create: if accesso() && request.resource.data.proprietario == io() && request.resource.data.usi == 0
        && request.resource.data.max is int
        && ((admin() && request.resource.data.max <= 1000)
          || (request.resource.data.max == 3
              && getAfter(/databases/$(database)/documents/utenti/$(io())).data.get('mioCodice', null) == codice
              && getAfter(/databases/$(database)/documents/utenti/$(io())).data.get('ammesso', false) == true));
      allow update: if accesso()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['usi'])
        && request.resource.data.usi == resource.data.usi + 1
        && request.resource.data.usi <= resource.data.max
        && resource.data.proprietario != io()
        && getAfter(/databases/$(database)/documents/utenti/$(io())).data.get('invitatoDa', null) == codice
        && getAfter(/databases/$(database)/documents/utenti/$(io())).data.get('ammesso', false) == true
        && !(exists(/databases/$(database)/documents/utenti/$(io()))
             && get(/databases/$(database)/documents/utenti/$(io())).data.get('ammesso', false) == true);
    }

    // Recensioni in due direzioni: una per noleggio concluso e per persona
    match /recensioni/{id} {
      allow read: if accesso();
      allow create: if scrittore() && recensioneOnesta(id);
      allow delete: if admin();

      function recensioneOnesta(id) {
        let d = request.resource.data;
        let n = get(/databases/$(database)/documents/noleggi/$(d.noleggio)).data;
        return d.da == io() && id == d.noleggio + '_' + io()
          && d.voto is int && d.voto >= 1 && d.voto <= 5
          && testo(d.testo, 300) && testo(d.daNome, 30)
          && d.quando == request.time
          && n.fase == 'CONCLUSO'
          && ((n.clienteUid == io() && d.a == n.proprietarioUid && d.ruoloA == 'proprietario')
            || (n.proprietarioUid == io() && d.a == n.clienteUid && d.ruoloA == 'cliente'));
      }
    }

    // Persone segnalate: ognuno conta una volta sola; gli admin riattivano
    match /segnalati/{uid} {
      allow read: if accesso() && (io() == uid || admin());
      allow list: if admin();
      allow create: if accesso() && io() != uid
        && request.resource.data.keys().hasOnly(['conteggio'])
        && request.resource.data.conteggio == 1
        && !exists(/databases/$(database)/documents/segnalati/$(uid)/da/$(io()))
        && existsAfter(/databases/$(database)/documents/segnalati/$(uid)/da/$(io()));
      allow update: if accesso() && (
        (io() != uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['conteggio'])
          && request.resource.data.conteggio == resource.data.get('conteggio', 0) + 1
          && !exists(/databases/$(database)/documents/segnalati/$(uid)/da/$(io()))
          && existsAfter(/databases/$(database)/documents/segnalati/$(uid)/da/$(io())))
        || (admin() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['conteggio', 'riattivato'])));
      match /da/{chi} {
        allow create: if accesso() && io() == chi && chi != uid;
        allow read: if admin();
      }
    }

    // Crash dell'app: chiunque sia dentro li invia, solo gli admin li leggono
    match /crash/{id} {
      allow create: if accesso() && request.resource.data.uid == io()
        && testo(request.resource.data.testo, 8000)
        && request.resource.data.keys().hasOnly(['testo', 'versione', 'telefono', 'android', 'leggero', 'uid', 'quando']);
      allow read, delete: if admin();
    }

    // Contatore anonimo delle visite al teaser: ognuno può solo aggiungere 1, solo gli admin leggono
    match /statistiche/{id} {
      function piuUno(k) {
        return request.resource.data.get(k, 0) == resource.data.get(k, 0)
          || request.resource.data.get(k, 0) == resource.data.get(k, 0) + 1;
      }
      allow read: if admin();
      allow create: if id == 'presto'
        && request.resource.data.keys().hasOnly(['totale', 'whatsapp', 'volantino', 'altro'])
        && request.resource.data.totale == 1
        && request.resource.data.keys().size() == 2;
      allow update: if id == 'presto'
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['totale', 'whatsapp', 'volantino', 'altro'])
        && request.resource.data.diff(resource.data).affectedKeys().size() == 2
        && request.resource.data.totale == resource.data.totale + 1
        && piuUno('whatsapp') && piuUno('volantino') && piuUno('altro');
    }

    // Badge: Pioniere (si attiva da solo se l'email era nella lista d'attesa), Fondatore e Ambasciatore (li danno gli admin)
    match /badge/{uid} {
      function soloBadge() {
        return request.resource.data.keys().hasOnly(['pioniere', 'fondatore', 'ambasciatore'])
          && request.resource.data.values().hasOnly([true, false]);
      }
      function pionierePermesso() {
        return accesso() && io() == uid
          && request.auth.token.email_verified == true
          && exists(/databases/$(database)/documents/attesa/$(request.auth.token.email.lower()))
          && request.resource.data.pioniere == true;
      }
      allow read: if accesso();
      allow create: if soloBadge() && (admin() || (pionierePermesso() && request.resource.data.keys().hasOnly(['pioniere'])));
      allow update: if soloBadge() && (admin() || (pionierePermesso() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['pioniere'])));
      allow delete: if admin();
    }

    // Lista d'attesa della pagina teaser: chiunque si iscrive (anche senza account), solo gli admin leggono
    match /attesa/{email} {
      allow create: if email.size() <= 100
        && email.matches('^[a-z0-9._%+-]+@[a-z0-9.-]+[.][a-z]{2,}$')
        && request.resource.data.keys().hasOnly(['email', 'quando'])
        && request.resource.data.email == email
        && request.resource.data.quando == request.time
        // i Pionieri si iscrivono solo prima del lancio (1 dicembre 2026)
        && request.time < timestamp.date(2026, 12, 1);
      allow read, delete: if admin();
    }

    // Foto profilo (separate, così la classifica resta leggera)
    match /foto_utenti/{uid} {
      allow read: if accesso();
      allow write: if accesso() && io() == uid && fotoOk(request.resource.data.get('foto', null));
      allow delete: if accesso() && io() == uid;
    }

    // Bici: tutti le vedono; solo il proprietario le modifica; voti uno a testa
    match /bici/{id} {
      allow read: if true;
      allow create: if scrittore() && biciValida()
        && request.resource.data.proprietarioUid == io()
        && request.resource.data.sommaVoti == 0 && request.resource.data.numVoti == 0
        && request.resource.data.rubata == false
        // una bici al minuto (l'ora la mette il server)
        && utenteDopo().get('ultimaBici', null) == request.time;
      allow update: if accesso() && !bandito() && (
        (resource.data.proprietarioUid == io()
          && cambiati().hasOnly(['disponibile', 'rubata', 'prezzo', 'soloFidati', 'proprietarioNome', 'cauzione', 'lucchetto', 'casco', 'orari', 'latRitiro', 'lonRitiro'])
          && (!('prezzo' in cambiati()) || (request.resource.data.prezzo is number
              && request.resource.data.prezzo >= 2 && request.resource.data.prezzo <= 15))
          && (!('cauzione' in cambiati()) || (request.resource.data.cauzione is number
              && request.resource.data.cauzione >= 30 && request.resource.data.cauzione <= 200))
          && testo(request.resource.data.proprietarioNome, 30))
        || votoOnesto()
      );
      allow delete: if (accesso() && resource.data.proprietarioUid == io()) || admin();

      function biciValida() {
        let d = request.resource.data;
        return testo(d.nome, 40) && d.nome.size() > 0
          && testo(d.descrizione, 600)
          && testo(d.proprietarioNome, 30)
          && d.prezzo is number && d.prezzo >= 2 && d.prezzo <= 15
          && d.tipo in ['CITY', 'MOUNTAIN', 'CORSA', 'EBIKE', 'MONOPATTINO']
          && d.get('cauzione', 30) is number && d.get('cauzione', 30) >= 30 && d.get('cauzione', 30) <= 200
          && d.get('lucchetto', false) == true
          && d.get('orari', []) is list && d.get('orari', []).size() <= 4
          && (d.get('latRitiro', null) == null || (d.latRitiro is number && d.latRitiro > 44.9 && d.latRitiro < 45.2))
          && (d.get('lonRitiro', null) == null || (d.lonRitiro is number && d.lonRitiro > 9.5 && d.lonRitiro < 9.9))
          && (d.get('prezzoMese', null) == null || (d.prezzoMese is number && d.prezzoMese > 0 && d.prezzoMese <= 1000))
          && fotoOk(d.get('foto', null));
      }
      function votoOnesto() {
        let voto = /databases/$(database)/documents/bici/$(id)/voti/$(io());
        return resource.data.proprietarioUid != io()
          && cambiati().hasOnly(['sommaVoti', 'numVoti'])
          && request.resource.data.numVoti == resource.data.numVoti + 1
          && !exists(voto)
          && getAfter(voto).data.voto == request.resource.data.sommaVoti - resource.data.sommaVoti;
      }

      match /voti/{uid} {
        allow read: if true;
        allow create: if scrittore() && io() == uid
          && request.resource.data.voto is int
          && request.resource.data.voto >= 1 && request.resource.data.voto <= 5;
      }
    }

    // Social: tutti vedono i giri; un like a testa
    match /post/{id} {
      allow read: if true;
      allow create: if scrittore()
        && request.resource.data.autoreUid == io()
        && request.resource.data.likes == 0
        && testo(request.resource.data.autoreNome, 30)
        && testo(request.resource.data.titolo, 60)
        && testo(request.resource.data.testo, 500)
        && fotoOk(request.resource.data.get('foto', null))
        // un post ogni 30 secondi
        && utenteDopo().get('ultimoPost', null) == request.time;
      allow update: if accesso() && !bandito() && likeOnesto();
      allow delete: if (accesso() && resource.data.autoreUid == io()) || admin();

      function likeOnesto() {
        let cuore = /databases/$(database)/documents/post/$(id)/like/$(io());
        return cambiati().hasOnly(['likes'])
          && ((request.resource.data.likes == resource.data.likes + 1 && !exists(cuore) && existsAfter(cuore))
           || (request.resource.data.likes == resource.data.likes - 1 && exists(cuore) && !existsAfter(cuore)));
      }

      match /like/{uid} {
        allow read: if true;
        allow create, delete: if accesso() && io() == uid;
      }
    }

    // Segnalazioni: tutti possono segnalare, solo gli admin le leggono
    match /segnalazioni/{id} {
      allow create: if accesso() && request.resource.data.daUid == io()
        && request.resource.data.tipo in ['post', 'bici']
        && testo(request.resource.data.doc, 60)
        && testo(request.resource.data.motivo, 300);
      allow read, delete: if admin();
    }

    // Noleggi: li vedono e li modificano solo chi noleggia e il proprietario
    match /noleggi/{id} {
      function partecipante() {
        return resource.data.clienteUid == io() || resource.data.proprietarioUid == io();
      }
      function partecipanteDi(noleggio) {
        let n = get(/databases/$(database)/documents/noleggi/$(noleggio)).data;
        return n.clienteUid == io() || n.proprietarioUid == io();
      }
      // il prezzo deve tornare con quello della bici (niente noleggi a 0 €)
      function prezzoOnesto() {
        let d = request.resource.data;
        let b = get(/databases/$(database)/documents/bici/$(d.biciDoc)).data;
        let ore = d.durataMs;
        let mese = b.get('prezzoMese', null);
        return d.prezzo is number && d.prezzo <= 1100
          && d.commissione is number && d.commissione >= 0.86
          && (
            (ore == 18000000 && d.prezzo >= b.prezzo)
            || (ore == 3600000 && d.prezzo >= 1.5 && d.prezzo >= b.prezzo * 0.4 - 0.26)
            || (ore == 7200000 && d.prezzo >= 2.5 && d.prezzo >= b.prezzo * 0.6 - 0.26)
            || (ore == 604800000 && mese != null && d.prezzo >= mese * 0.35 - 0.26)
            || (ore == 2592000000 && mese != null && d.prezzo >= mese)
          );
      }
      allow read: if accesso() && (partecipante() || admin());
      // gli admin decidono le liti sui danni (solo stato e importo)
      allow update: if admin() && resource.data.statoDanno == 3
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['statoDanno', 'importoDanno'])
        && request.resource.data.statoDanno == 4;
      allow create: if scrittore() && request.resource.data.clienteUid == io()
        && request.resource.data.fase == 'IN_ATTESA'
        && request.resource.data.codiceOk == false
        // il proprietario è davvero quello della bici, e non puoi noleggiare la tua
        && request.resource.data.proprietarioUid == get(/databases/$(database)/documents/bici/$(request.resource.data.biciDoc)).data.proprietarioUid
        && request.resource.data.proprietarioUid != io()
        && testo(request.resource.data.clienteNome, 30)
        && prezzoOnesto();
      allow update: if accesso() && !bandito() && partecipante()
        // nessuno può cambiare chi sono il cliente e il proprietario, né la bici
        && request.resource.data.clienteUid == resource.data.clienteUid
        && request.resource.data.proprietarioUid == resource.data.proprietarioUid
        && request.resource.data.biciDoc == resource.data.biciDoc
        // solo il proprietario conferma il codice di consegna
        && (resource.data.proprietarioUid == io()
            || !cambiati().hasAny(['codiceOk']))
        // si parte solo dopo il codice giusto
        && (request.resource.data.fase != 'IN_CORSO' || request.resource.data.codiceOk == true);

      // Chat del noleggio
      match /messaggi/{m} {
        allow read: if accesso() && partecipanteDi(id);
        allow create: if scrittore() && partecipanteDi(id)
          && request.resource.data.daUid == io()
          && testo(request.resource.data.testo, 1000);
      }

      // Foto del ritiro e della restituzione: la prova nelle liti
      match /foto/{tipo} {
        allow read: if accesso() && (partecipanteDi(id) || admin());
        allow create, update: if accesso() && partecipanteDi(id) && !bandito()
          && tipo in ['consegna', 'restituzione']
          && request.resource.data.da == io()
          && fotoOk(request.resource.data.foto) && request.resource.data.foto != null;
      }

      // Codice di consegna segreto: lo vede solo il proprietario
      match /privato/{p} {
        allow read, write: if accesso()
          && get(/databases/$(database)/documents/noleggi/$(id)).data.proprietarioUid == io();
      }
    }
  }
}
